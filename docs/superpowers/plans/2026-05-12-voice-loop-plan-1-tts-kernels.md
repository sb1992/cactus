# Voice Loop — Plan 1: TTS Kernel Foundation

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Land the kernel ops cactus needs to run Kokoro-82M TTS — FFT/iFFT, STFT/ISTFT, LSTM cell, transposed conv1d — each with passing correctness tests against scipy/PyTorch references.

**Architecture:** Vendor PFFFT (single-file, BSD-3, NEON-optimized) for the FFT primitive. Wrap as a cactus FFT op. Build STFT/ISTFT on top using cactus's existing windowing patterns. LSTM and transposed conv1d are written as new kernel ops following cactus's existing NEON kernel style. Every op gets a unit test that compares its output to a Python reference (scipy for FFT, torch for LSTM/conv).

**Tech Stack:** C++17, ARM NEON intrinsics, GoogleTest, CMake, vendored PFFFT. Reference Python validation uses scipy + torch.

**Spec:** [`docs/superpowers/specs/2026-05-12-voice-loop-stage-1-design.md`](../specs/2026-05-12-voice-loop-stage-1-design.md)

**Definition of Done:** All four new ops (`cactus_fft`, `cactus_istft`, `cactus_lstm_cell`, `cactus_conv1d_transpose`) have passing unit tests in CI. The full cactus test suite passes. PFFFT is vendored with its license. No regressions in existing tests.

```cpp
TEST(PFFFTSmoke, RoundTrip256RealFFT) {
    constexpr int N = 256;
    PFFFT_Setup* setup = pffft_new_setup(N, PFFFT_REAL);
    ASSERT_NE(setup, nullptr);

    std::vector<float> input(N);
    std::vector<float> spectrum(N);
    std::vector<float> output(N);

    for (int i = 0; i < N; ++i) {
        input[i] = std::sin(2.0f * M_PI * 5.0f * i / N);  // 5-cycle sine

    pffft_transform_ordered(setup, spectrum.data(), output.data(), nullptr, PFFFT_BACKWARD);

    for (int i = 0; i < N; ++i) {

    }

    for (int i = 0; i < N; ++i) {
        EXPECT_NEAR(input[i], output[i], 1e-5f) << "mismatch at index " << i;
    }

    pffft_destroy_setup(setup);
}
```

```cpp
#pragma once

#include <cstddef>

// Real-to-complex forward FFT.
//   input:  pointer to N float samples (time domain)
//   output: pointer to (N/2 + 1) * 2 floats (interleaved re/im in scipy.fft.rfft order:
//           [re0, im0, re1, im1, ..., re_{N/2}, im_{N/2}]; im0 and im_{N/2} are always 0)
//   n:      transform size; must be a multiple of 32 and >= 32 for PFFFT
void cactus_fft_r2c(const float* input, float* output, size_t n);

// Complex-to-real inverse FFT (reverse of cactus_fft_r2c).
//   input:  (N/2 + 1) * 2 floats interleaved re/im
//   output: N floats (time domain), normalized (divided by N internally)
//   n:      transform size
void cactus_fft_c2r(const float* input, float* output, size_t n);
```

- [ ] **Step 5: Write the implementation**

Create `cactus/kernel/kernel_fft.cpp`:

```cpp
#include "kernel_fft.h"
#include "pffft.h"
#include <stdexcept>
#include <cstring>
#include <vector>

// PFFFT layout note:
//   PFFFT_REAL forward output is "ordered" only when using
//   pffft_transform_ordered. The ordered layout for an N-point real FFT is
//   exactly the same as scipy's rfft: [re0, re_{N/2}, re1, im1, re2, im2, ...]
//   PFFFT packs re_{N/2} into slot[1] (the "Nyquist" slot) which scipy puts
//   separately at the end. We unpack to scipy's layout.

void cactus_fft_r2c(const float* input, float* output, size_t n) {
    if (n < 32 || (n % 32) != 0) {
        throw std::runtime_error("cactus_fft_r2c: n must be >= 32 and a multiple of 32");
    }

    PFFFT_Setup* setup = pffft_new_setup(static_cast<int>(n), PFFFT_REAL);
    if (!setup) throw std::runtime_error("pffft_new_setup failed");

    std::vector<float> tmp(n);
    pffft_transform_ordered(setup, input, tmp.data(), nullptr, PFFFT_FORWARD);

    // Unpack PFFFT layout → scipy layout.
    // PFFFT ordered: tmp[0]=re0, tmp[1]=re_{N/2}, tmp[2]=re1, tmp[3]=im1, ...
    // scipy:         re0, im0=0, re1, im1, ..., re_{N/2}, im_{N/2}=0
    output[0] = tmp[0];          // re0
    output[1] = 0.0f;            // im0
    for (size_t k = 1; k < n / 2; ++k) {
        output[2 * k]     = tmp[2 * k];      // re_k
        output[2 * k + 1] = tmp[2 * k + 1];  // im_k
    }
    output[n]     = tmp[1];      // re_{N/2}
    output[n + 1] = 0.0f;        // im_{N/2}

    pffft_destroy_setup(setup);
}

void cactus_fft_c2r(const float* input, float* output, size_t n) {
    if (n < 32 || (n % 32) != 0) {
        throw std::runtime_error("cactus_fft_c2r: n must be >= 32 and a multiple of 32");
    }

    PFFFT_Setup* setup = pffft_new_setup(static_cast<int>(n), PFFFT_REAL);
    if (!setup) throw std::runtime_error("pffft_new_setup failed");

    // Pack scipy layout → PFFFT layout
    std::vector<float> tmp(n);
    tmp[0] = input[0];           // re0
    tmp[1] = input[n];           // re_{N/2}
    for (size_t k = 1; k < n / 2; ++k) {
        tmp[2 * k]     = input[2 * k];
        tmp[2 * k + 1] = input[2 * k + 1];
    }

    pffft_transform_ordered(setup, tmp.data(), output, nullptr, PFFFT_BACKWARD);

    const float scale = 1.0f / static_cast<float>(n);
    for (size_t i = 0; i < n; ++i) output[i] *= scale;

    pffft_destroy_setup(setup);
}
```

```python
from scipy.signal import stft, istft

# Test 4: STFT/ISTFT round-trip on a chirp signal
fs = 24000
duration = 0.5
N = int(fs * duration)
t = np.arange(N) / fs
chirp = np.sin(2 * np.pi * (200 + 1000 * t) * t).astype(np.float32)

n_fft = 512
hop = 128

f, tt, Zxx = stft(chirp, fs=fs, nperseg=n_fft, noverlap=n_fft - hop,
                  window='hann', boundary=None, padded=False)
# Zxx shape: (n_fft//2+1, n_frames), complex64
write_f32(os.path.join(OUT, "stft_input_chirp.bin"), chirp)

# Save Zxx as interleaved [re, im] per frame, frame-major:
# layout: n_frames * (n_fft//2+1) * 2 floats
n_freq = n_fft // 2 + 1
```

```cpp
#pragma once

// Short-Time Fourier Transform with Hann window.
//   input:    pointer to n_samples float PCM samples
//   output:   pointer to n_frames * (n_fft/2 + 1) * 2 floats (interleaved re/im, frame-major)
//   n_fft:    window/FFT size (must be valid for PFFFT — multiple of 32, >= 32)
//   hop:      hop size in samples
//   n_frames: number of frames; should equal (n_samples - n_fft) / hop + 1
void cactus_stft(const float* input, int n_samples,
                 float* output, int n_fft, int hop, int n_frames);

// Inverse STFT with Hann window + overlap-add + window-sum normalization.
//   input:    pointer to n_frames * (n_fft/2 + 1) * 2 floats (scipy rfft interleaved layout)
//   output:   pointer to n_samples floats; caller must zero-init before the call
//   The output length must satisfy n_samples >= (n_frames - 1) * hop + n_fft
void cactus_istft(const float* input, int n_frames,
                  float* output, int n_samples,
                  int n_fft, int hop);
```

```cpp
#include "kernel_stft.h"
#include "kernel_fft.h"
#include <vector>
#include <cmath>
#include <cstring>

static std::vector<float> hann_window(int n) {
    std::vector<float> w(n);
    for (int i = 0; i < n; ++i) {
        w[i] = 0.5f * (1.0f - std::cos(2.0f * 3.14159265358979323846f * i / (n - 1)));
    }
    return w;
}

void cactus_stft(const float* input, int n_samples,
                 float* output, int n_fft, int hop, int n_frames) {
    auto window = hann_window(n_fft);
    std::vector<float> frame(n_fft);
    const int n_freq = n_fft / 2 + 1;

    for (int f = 0; f < n_frames; ++f) {
        const int start = f * hop;
        for (int i = 0; i < n_fft; ++i) {
            frame[i] = (start + i < n_samples) ? input[start + i] * window[i] : 0.0f;
        }
        cactus_fft_r2c(frame.data(), output + f * n_freq * 2, n_fft);
    }
}

void cactus_istft(const float* input, int n_frames,
                  float* output, int n_samples,
                  int n_fft, int hop) {
    auto window = hann_window(n_fft);
    std::vector<float> frame(n_fft);
    std::vector<float> norm(n_samples, 0.0f);
    const int n_freq = n_fft / 2 + 1;

    std::memset(output, 0, n_samples * sizeof(float));

    for (int f = 0; f < n_frames; ++f) {
        cactus_fft_c2r(input + f * n_freq * 2, frame.data(), n_fft);
        const int start = f * hop;
        for (int i = 0; i < n_fft; ++i) {
            const int idx = start + i;
            if (idx < n_samples) {
                output[idx] += frame[i] * window[i];
                norm[idx] += window[i] * window[i];
            }
        }
    }

    // Normalize by window-sum to invert the synthesis windowing
    for (int i = 0; i < n_samples; ++i) {
        if (norm[i] > 1e-8f) output[i] /= norm[i];
    }
}
```

```cpp
#pragma once

// PyTorch-compatible LSTM cell forward.
//
// Inputs (all row-major, float32):
//   x:    [I]      input
//   h_in: [H]      previous hidden state
//   c_in: [H]      previous cell state
//   W_ih: [4H, I]  input-hidden weights (gates ordered: i, f, g, o per PyTorch)
//   W_hh: [4H, H]  hidden-hidden weights
//   b_ih: [4H]     input-hidden bias
//   b_hh: [4H]     hidden-hidden bias
//
// Outputs:
//   h_out: [H]
//   c_out: [H]
//
// Math (per PyTorch nn.LSTMCell):
//   gates = W_ih @ x + b_ih + W_hh @ h_in + b_hh           // [4H]
//   i, f, g, o = split(gates, H)                            // each [H]
//   c_out = sigmoid(f) * c_in + sigmoid(i) * tanh(g)
//   h_out = sigmoid(o) * tanh(c_out)
void cactus_lstm_cell(
    const float* x, const float* h_in, const float* c_in,
    const float* W_ih, const float* W_hh,
    const float* b_ih, const float* b_hh,
    float* h_out, float* c_out,
    int I, int H);
```

- [ ] **Step 5: Write the implementation**

Create `cactus/kernel/kernel_lstm.cpp`:

```cpp
#include "kernel_lstm.h"
#include <cmath>
#include <vector>

static inline float sigmoidf(float x) {
    return 1.0f / (1.0f + std::exp(-x));
}

void cactus_lstm_cell(
    const float* x, const float* h_in, const float* c_in,
    const float* W_ih, const float* W_hh,
    const float* b_ih, const float* b_hh,
    float* h_out, float* c_out,
    int I, int H)
{
    // gates = W_ih @ x + b_ih + W_hh @ h_in + b_hh
    // (This is the reference scalar implementation. NEON tiling can be added
    //  later by replacing the inner accumulation loops with cactus_gemv calls
    //  if the kernel pool supports them — see cactus/kernel/kernel.cpp for the
    //  cactus_gemv_int4_block_range pattern.)
    std::vector<float> gates(4 * H);
    for (int g = 0; g < 4 * H; ++g) {
        float acc = b_ih[g] + b_hh[g];
        // W_ih row g, dot with x
        const float* w_ih_row = W_ih + g * I;
        for (int i = 0; i < I; ++i) acc += w_ih_row[i] * x[i];
        // W_hh row g, dot with h_in
        const float* w_hh_row = W_hh + g * H;
        for (int j = 0; j < H; ++j) acc += w_hh_row[j] * h_in[j];
        gates[g] = acc;
    }

    // PyTorch gate order: [i, f, g, o] each of length H
    const float* gi = gates.data() + 0 * H;
    const float* gf = gates.data() + 1 * H;
    const float* gg = gates.data() + 2 * H;
    const float* go = gates.data() + 3 * H;

    for (int j = 0; j < H; ++j) {
        const float i_t = sigmoidf(gi[j]);
        const float f_t = sigmoidf(gf[j]);
        const float g_t = std::tanh(gg[j]);
        const float o_t = sigmoidf(go[j]);
        c_out[j] = f_t * c_in[j] + i_t * g_t;
        h_out[j] = o_t * std::tanh(c_out[j]);
    }
}
```

```cpp
void cactus_conv1d_transpose(
    const float* x, const float* W, const float* b, float* y,
    int c_in, int c_out, int t_in, int k, int stride, int padding)
{
    const int t_out = (t_in - 1) * stride - 2 * padding + k;
    std::memset(y, 0, c_out * t_out * sizeof(float));

    // Scatter: for each (ic, t_in, kk) input position+kernel-tap, add into output at
    //   t_pos = t_in * stride + kk - padding
    for (int ic = 0; ic < c_in; ++ic) {
        for (int ti = 0; ti < t_in; ++ti) {
            const float xv = x[ic * t_in + ti];
            for (int oc = 0; oc < c_out; ++oc) {
                const float* w_ic_oc = W + (ic * c_out + oc) * k;
                float* y_oc = y + oc * t_out;
                for (int kk = 0; kk < k; ++kk) {
                    const int t_pos = ti * stride + kk - padding;
                    if (t_pos >= 0 && t_pos < t_out) {
                        y_oc[t_pos] += xv * w_ic_oc[kk];
                    }
                }
            }
        }
    }

    if (b) {
        for (int oc = 0; oc < c_out; ++oc) {
            float* y_oc = y + oc * t_out;
            const float bv = b[oc];
            for (int t = 0; t < t_out; ++t) y_oc[t] += bv;
        }
    }
}
```
